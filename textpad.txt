---------- UFS command RAM-log capture ----------

RAMLOG_CMD=${RAMLOG_CMD:-/data/ufs_cmd_log}
RAMLOG_PID=""
RAMLOG_FILE=""

ramlog_start() {
[ -x "$RAMLOG_CMD" ] || {
log "ERROR: UFS command logger not executable: $RAMLOG_CMD"
return 1
}

mkdir -p "$RES_DIR/ramlogs"
RAMLOG_FILE="$RES_DIR/ramlogs/${TEST_NAME}.ramlog"

log "Clearing UFS RAM-log buffer: $TEST_NAME"
"$RAMLOG_CMD" -c || {
    log "ERROR: Failed to clear UFS RAM-log buffer"
    return 1
}

log "Starting UFS RAM-log capture in background"

"$RAMLOG_CMD" -s >/dev/null 2>&1 &
RAMLOG_PID=$!

# Give the application a moment to initialize.
sleep 1

if ! kill -0 "$RAMLOG_PID" 2>/dev/null; then
    wait "$RAMLOG_PID" 2>/dev/null
    RAMLOG_PID=""
    log "ERROR: UFS RAM-log capture process exited unexpectedly"
    return 1
fi

return 0

}

ramlog_stop() {
[ -n "$RAMLOG_PID" ] || return 0

log "Stopping UFS RAM-log capture"

# Request the same graceful termination as Ctrl+C.
kill -INT "$RAMLOG_PID" 2>/dev/null

# Wait for the capture process to finish.
wait "$RAMLOG_PID" 2>/dev/null
RAMLOG_PID=""

log "Saving UFS RAM-log: $RAMLOG_FILE"

if "$RAMLOG_CMD" -l "$RAMLOG_FILE"; then
    log "UFS RAM-log saved: $RAMLOG_FILE"
else
    log "ERROR: Failed to save UFS RAM-log: $RAMLOG_FILE"
    TEST_STATUS=FAIL
    return 1
fi

return 0

}
