#!/bin/bash

JOB_DIR=./jobs
LOG_DIR=./LOGS
FIO=fio
job_ops=""
job_title=""
#MKFS_F2FS=../f2fs_tools_bins/inline_tail/
MKFS_F2FS=""


CNT=$1
loop=$CNT
slp_time=10


create_jobs()
{
        rm ${JOB_DIR}/*
        CYC=$2 # Pass the cycle count as the second argument

        if [ ${1} -eq 0 ]; then
                echo ""
                echo "************************************************************************"
                echo "choose operation for job file creation"
                echo "************************************************************************"
                echo "  1. Sequential Write"
                read OPS
        else
                OPS=$1
        fi

        case $OPS in
                1) job_ops="write"; job_title="seq_write" ;;
                3) job_ops="randwrite"; job_title="rand_write" ;;
                *) job_ops="write"; job_title="seq_write" ;;
        esac

        # Notice the added filename_format utilizing the $CYC variable
        for sz in 3_5k:35k 5k:50k 63k:630k 64k:640k 65k:650k 70k:700k 73k:730k; do
                name=$(echo $sz | cut -d: -f1)
                bytes=$(echo $sz | cut -d: -f2)
                
                cat <<EOF > ${JOB_DIR}/${name}_${job_title}.fio
[global]
ioengine=io_uring
iodepth=1
direct=0
directory=/mnt/lun0
openfiles=10
nrfiles=10
filename_format=job_${name}_\$filenum_cycle_${CYC}

[${name}_${job_title}]
rw=${job_ops}
bs=1k
size=${bytes}
EOF
        done
}


exec_fio()
{
        echo "# ------------------------------------------------------------------------"
        for wr in 3_5k 5k 63k 64k 65k 70k 73k; do
                ${FIO} ${JOB_DIR}/${wr}_${job_title}.fio | tee ${LOG_DIR}/${wr}_${job_title}.log
        done

        echo "# ------------------------------------------------------------------------"

<<'####'
        if [ ! -d ${LOG_DIR} ]; then
                mkdir -p ${LOG_DIR}
        fi

        cp ${LOG_DIR}/*.log ${LOG_DIR}/${MNT_OPS}/
        rm -rf ${LOG_DIR}/*.log
        #rm -rf ${JOB_DIR}/*
####

}

remove_files()
{
        rm -rf /mnt/lun0/*
        echo "# Removed previous test"
}

init_dev()
{
        umount /dev/sda1
        #mkfs.f2fs -f /dev/sda1
	mkfs.f2fs -f -O extra_attr,compression /dev/sda1
        #mount -t f2fs /dev/sda1 /mnt/lun0 -o mode=lfs,alloc_mode=default
	mount -t f2fs /dev/sda1 /mnt/lun0 -o mode=lfs,alloc_mode=default,compress_extension=*

}

echo ""
echo "************************************************************************"
init_dev
echo "************************************************************************"
echo -n "Test loop count? : "
read CNT

loop=$CNT


# Test loop
# Update your main loop at the bottom of the script to pass $cycle
while [ $loop -gt 0 ]
do
        cycle=$(((CNT-loop)+1))
        echo "# ========================================================================= "
        echo "# TEST_CYCLE: $cycle"
        echo "# ========================================================================= "
        create_jobs 1 $cycle  # <--- Changed this line to pass $cycle
        exec_fio
        loop=$((loop-1))
done

echo "# ------------------------------------------------------------------------"
echo "# Test Completed"
