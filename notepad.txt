while true; do
    echo "========== $(date) =========="

    echo -n "DF: "
    df -h /mnt/pass_through/0/emulated | tail -1

    echo -n "free_segments: "
    cat /sys/fs/f2fs/sda11/free_segments

    echo -n "dirty_segments: "
    cat /sys/fs/f2fs/sda11/dirty_segments

    echo -n "FGGC: "
    cat /sys/fs/f2fs/sda11/gc_foreground_calls

    echo -n "BGGC: "
    cat /sys/fs/f2fs/sda11/gc_background_calls

    echo -n "moved FG: "
    cat /sys/fs/f2fs/sda11/moved_blocks_foreground

    echo -n "moved BG: "
    cat /sys/fs/f2fs/sda11/moved_blocks_background

    echo -n "reclaimed: "
    cat /sys/fs/f2fs/sda11/gc_reclaimed_segments

    echo
    sleep 1
done
