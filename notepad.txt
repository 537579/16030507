static int move_data_block(struct inode *inode, block_t bidx,
				int gc_type, unsigned int segno, int off)
{
	struct address_space *mapping = inode->i_mapping;
	struct inode *atomic_inode = NULL;
	struct f2fs_io_info fio = {
		.sbi = F2FS_I_SB(inode),
		.ino = inode->i_ino,
		.type = DATA,
		.temp = COLD,
		.op = REQ_OP_READ,
		.op_flags = 0,
		.encrypted_page = NULL,
		.in_list = 0,
	};
	struct dnode_of_data dn;
	struct f2fs_summary sum;
	struct node_info ni;
	struct folio *folio, *mfolio, *efolio;
	block_t newaddr;
	int err = 0;
	bool lfs_mode = f2fs_lfs_mode(fio.sbi);
	bool copy_offloaded = false;
	u64 flow_id;

	int type = fio.sbi->am.atgc_enabled &&
			(gc_type == BG_GC) &&
			(fio.sbi->gc_mode != GC_URGENT_HIGH) ?
			CURSEG_ALL_DATA_ATGC :
			CURSEG_COLD_DATA;

	/*
	 * FLOW ID identifies this GC block migration.
	 *
	 * victim segment + block offset uniquely identifies
	 * the source block within one GC migration context.
	 */
	flow_id = ((u64)segno << 32) | (u32)off;

	/*
	 * Handle COW / atomic files.
	 */
	f2fs_down_read(&F2FS_I(inode)->i_sem);

	if (f2fs_is_cow_file(inode)) {
		atomic_inode = igrab(F2FS_I(inode)->atomic_inode);

		if (!atomic_inode) {
			f2fs_up_read(&F2FS_I(inode)->i_sem);
			return -EBUSY;
		}

		mapping = atomic_inode->i_mapping;
	}

	f2fs_up_read(&F2FS_I(inode)->i_sem);

	/*
	 * Grab destination page cache folio.
	 *
	 * Do not read the page here.
	 */
	folio = f2fs_grab_cache_folio(mapping, bidx, false);

	if (IS_ERR(folio)) {
		err = PTR_ERR(folio);
		goto out_iput;
	}

	/*
	 * Verify that this block is still valid in the victim segment.
	 */
	if (!check_valid_map(F2FS_I_SB(inode), segno, off)) {

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"BLOCK_INVALID_SKIP "
			"victim_seg=%u off=%d",
			(unsigned long long)flow_id,
			segno, off);

		err = -ENOENT;
		goto out;
	}

	f2fs_info(fio.sbi,
		"[F2FS_GC][FLOW=%016llx] "
		"MIGRATION_START "
		"victim_seg=%u off=%d ino=%lu bidx=%llu",
		(unsigned long long)flow_id,
		segno,
		off,
		inode->i_ino,
		(unsigned long long)bidx);

	/*
	 * Check whether this inode/block can be moved by GC.
	 */
	err = f2fs_gc_pinned_control(inode, gc_type, segno);

	if (err) {

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"MIGRATION_SKIP_PINNED err=%d",
			(unsigned long long)flow_id,
			err);

		goto out;
	}

	set_new_dnode(&dn, inode, NULL, NULL, 0);

	err = f2fs_get_dnode_of_data(&dn,
					bidx,
					LOOKUP_NODE);

	if (err)
		goto out;

	/*
	 * Block may have been invalidated after initial victim
	 * segment validity check.
	 */
	if (unlikely(dn.data_blkaddr == NULL_ADDR)) {

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"BLOCK_MAPPING_INVALID",
			(unsigned long long)flow_id);

		folio_clear_uptodate(folio);

		err = -ENOENT;
		goto put_out;
	}

	/*
	 * Ensure any previous dirty/writeback activity has completed.
	 */
	f2fs_folio_wait_writeback(folio,
					DATA,
					true,
					true);

	f2fs_wait_on_block_writeback(inode,
					dn.data_blkaddr);

	err = f2fs_get_node_info(fio.sbi,
					dn.nid,
					&ni,
					false);

	if (err)
		goto put_out;

	/*
	 * SOURCE BLOCK SETUP
	 */
	fio.folio = folio;

	fio.old_blkaddr = dn.data_blkaddr;
	fio.new_blkaddr = dn.data_blkaddr;

	/*
	 * LFS ordering protection.
	 */
	if (lfs_mode)
		f2fs_down_write(&fio.sbi->io_order_lock);

	/*
	 * ============================================================
	 * STEP 1: READ SOURCE BEFORE DESTINATION ALLOCATION
	 * ============================================================
	 *
	 * This ordering is intentional.
	 *
	 * Source block is read into META_MAPPING before destination
	 * allocation changes F2FS SIT/segment state.
	 */
	mfolio = f2fs_grab_cache_folio(META_MAPPING(fio.sbi),
					fio.old_blkaddr,
					false);

	if (IS_ERR(mfolio)) {
		err = PTR_ERR(mfolio);
		goto up_out;
	}

	fio.encrypted_page =
		folio_file_page(mfolio,
				fio.old_blkaddr);

	/*
	 * Read source physical block.
	 */
	if (!folio_test_uptodate(mfolio)) {

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"SOURCE_READ "
			"SRC_PBA=0x%llx",
			(unsigned long long)flow_id,
			(unsigned long long)fio.old_blkaddr);

		err = f2fs_submit_page_bio(&fio);

		if (err) {
			f2fs_folio_put(mfolio, true);
			goto up_out;
		}

		f2fs_update_iostat(fio.sbi,
					inode,
					FS_DATA_READ_IO,
					F2FS_BLKSIZE);

		f2fs_update_iostat(fio.sbi,
					NULL,
					FS_GDATA_READ_IO,
					F2FS_BLKSIZE);

		folio_lock(mfolio);

		if (unlikely(!is_meta_folio(mfolio) ||
			     !folio_test_uptodate(mfolio))) {

			err = -EIO;

			f2fs_folio_put(mfolio, true);
			goto up_out;
		}
	}

	/*
	 * ============================================================
	 * STEP 2: ALLOCATE DESTINATION
	 * ============================================================
	 */
	set_summary(&sum,
		    dn.nid,
		    dn.ofs_in_node,
		    ni.version);

	err = f2fs_allocate_data_block(fio.sbi,
					NULL,
					fio.old_blkaddr,
					&newaddr,
					&sum,
					type,
					NULL);

	if (err) {

		f2fs_folio_put(mfolio, true);

		goto up_out;
	}

	fio.new_blkaddr = newaddr;

	f2fs_info(fio.sbi,
		"[F2FS_GC][FLOW=%016llx] "
		"DEST_ALLOCATED "
		"SRC_PBA=0x%llx "
		"DST_PBA=0x%llx "
		"DST_SEG=%u",
		(unsigned long long)flow_id,
		(unsigned long long)fio.old_blkaddr,
		(unsigned long long)newaddr,
		GET_SEGNO(fio.sbi, newaddr));

	/*
	 * ============================================================
	 * STEP 3: TRY NATIVE COPY OFFLOAD
	 * ============================================================
	 *
	 * The helper should:
	 *
	 *   - resolve F2FS PBA → block device + sector
	 *   - construct blk_copy_seg input/output
	 *   - check max_copy_sectors
	 *   - call blkdev_copy_offload()
	 *
	 * Success:
	 *      destination block already contains source data
	 *
	 * Failure:
	 *      continue existing CPU copy path
	 */
	err = f2fs_gc_try_copy_offload(&fio);

	if (!err) {

		copy_offloaded = true;

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"COPY_OFFLOAD_SUCCESS "
			"SRC_PBA=0x%llx "
			"DST_PBA=0x%llx "
			"METHOD=DEVICE_COPY",
			(unsigned long long)flow_id,
			(unsigned long long)fio.old_blkaddr,
			(unsigned long long)newaddr);

	} else {

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"COPY_OFFLOAD_FAILED "
			"SRC_PBA=0x%llx "
			"DST_PBA=0x%llx "
			"ERR=%d "
			"FALLBACK=CPU_COPY",
			(unsigned long long)flow_id,
			(unsigned long long)fio.old_blkaddr,
			(unsigned long long)newaddr,
			err);

		/*
		 * Do not propagate copy-offload errors.
		 *
		 * Existing CPU migration remains the fallback.
		 */
		err = 0;
	}

	/*
	 * ============================================================
	 * STEP 4A: NATIVE COPY SUCCESS
	 * ============================================================
	 */
	if (copy_offloaded) {

		/*
		 * Source data has already been copied to newaddr.
		 *
		 * No need to:
		 *
		 *   - allocate efolio
		 *   - memcpy()
		 *   - submit REQ_OP_WRITE
		 */
		f2fs_folio_put(mfolio, true);

		/*
		 * Maintain F2FS cache coherency.
		 */
		f2fs_invalidate_internal_cache(fio.sbi,
						fio.old_blkaddr,
						1);

		/*
		 * Commit new physical mapping.
		 */
		f2fs_update_data_blkaddr(&dn,
					 newaddr);

		set_inode_flag(inode,
			       FI_APPEND_WRITE);

		f2fs_update_iostat(fio.sbi,
					NULL,
					FS_GC_DATA_IO,
					F2FS_BLKSIZE);

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"MAPPING_UPDATED "
			"SRC_PBA=0x%llx "
			"DST_PBA=0x%llx "
			"RESULT=OFFLOAD",
			(unsigned long long)flow_id,
			(unsigned long long)fio.old_blkaddr,
			(unsigned long long)newaddr);

		goto put_out;
	}

	/*
	 * ============================================================
	 * STEP 4B: ORIGINAL CPU COPY FALLBACK
	 * ============================================================
	 */
	f2fs_info(fio.sbi,
		"[F2FS_GC][FLOW=%016llx] "
		"CPU_COPY_START "
		"SRC_PBA=0x%llx "
		"DST_PBA=0x%llx",
		(unsigned long long)flow_id,
		(unsigned long long)fio.old_blkaddr,
		(unsigned long long)newaddr);

	/*
	 * Create destination folio.
	 */
	efolio = f2fs_filemap_get_folio(
			META_MAPPING(fio.sbi),
			newaddr,
			FGP_LOCK | FGP_CREAT,
			GFP_NOFS);

	if (IS_ERR(efolio)) {

		err = PTR_ERR(efolio);

		f2fs_folio_put(mfolio, true);

		goto recover_block;
	}

	fio.encrypted_page = &efolio->page;

	/*
	 * ============================================================
	 * ORIGINAL CPU memcpy() PATH
	 * ============================================================
	 */
	f2fs_wait_on_page_writeback(
			fio.encrypted_page,
			DATA,
			true,
			true);

	memcpy(page_address(fio.encrypted_page),
	       folio_address(mfolio),
	       PAGE_SIZE);

	f2fs_folio_put(mfolio, true);

	f2fs_invalidate_internal_cache(fio.sbi,
					fio.old_blkaddr,
					1);

	set_page_dirty(fio.encrypted_page);

	if (clear_page_dirty_for_io(
			fio.encrypted_page))
		dec_page_count(fio.sbi,
				F2FS_DIRTY_META);

	set_page_writeback(fio.encrypted_page);

	/*
	 * Normal CPU fallback write.
	 */
	fio.op = REQ_OP_WRITE;

	fio.op_flags = REQ_SYNC;

	fio.new_blkaddr = newaddr;

	f2fs_submit_page_write(&fio);

	f2fs_update_iostat(fio.sbi,
				NULL,
				FS_GC_DATA_IO,
				F2FS_BLKSIZE);

	/*
	 * Commit mapping only after the fallback write has been
	 * submitted, preserving the original implementation ordering.
	 */
	f2fs_update_data_blkaddr(&dn,
				 newaddr);

	set_inode_flag(inode,
		       FI_APPEND_WRITE);

	f2fs_info(fio.sbi,
		"[F2FS_GC][FLOW=%016llx] "
		"MAPPING_UPDATED "
		"SRC_PBA=0x%llx "
		"DST_PBA=0x%llx "
		"RESULT=CPU_COPY",
		(unsigned long long)flow_id,
		(unsigned long long)fio.old_blkaddr,
		(unsigned long long)newaddr);

	f2fs_put_page(fio.encrypted_page,
		      true);

	goto put_out;

	/*
	 * ============================================================
	 * ERROR RECOVERY
	 * ============================================================
	 */
recover_block:

	if (err) {

		f2fs_info(fio.sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"MIGRATION_RECOVER "
			"SRC_PBA=0x%llx "
			"DST_PBA=0x%llx "
			"ERR=%d",
			(unsigned long long)flow_id,
			(unsigned long long)fio.old_blkaddr,
			(unsigned long long)newaddr,
			err);

		f2fs_do_replace_block(fio.sbi,
				      &sum,
				      newaddr,
				      fio.old_blkaddr,
				      true,
				      true,
				      true);
	}

up_out:

	if (lfs_mode)
		f2fs_up_write(&fio.sbi->io_order_lock);

put_out:
	f2fs_put_dnode(&dn);

out:
	f2fs_folio_put(folio, true);

out_iput:

	if (atomic_inode)
		iput(atomic_inode);

	return err;
}

static int f2fs_gc_try_copy_offload(struct f2fs_sb_info *sbi,
				    block_t old_blkaddr,
				    block_t new_blkaddr,
				    u64 flow_id)
{
	struct blk_copy_seg blk_copy_in_seg;
	struct blk_copy_seg blk_copy_out_seg;
	struct blk_copy_params params = { };
	struct block_device *src_bdev;
	struct block_device *dst_bdev;
	sector_t src_sector;
	sector_t dst_sector;
	int ret;

	src_bdev = f2fs_target_device(sbi,
				      old_blkaddr,
				      &src_sector);

	dst_bdev = f2fs_target_device(sbi,
				      new_blkaddr,
				      &dst_sector);

	blk_copy_in_seg.pos =
		(loff_t)src_sector << SECTOR_SHIFT;

	blk_copy_in_seg.len =
		F2FS_BLKSIZE;

	blk_copy_out_seg.pos =
		(loff_t)dst_sector << SECTOR_SHIFT;

	blk_copy_out_seg.len =
		F2FS_BLKSIZE;

	params.in_bdev = src_bdev;
	params.in_segs = &blk_copy_in_seg;
	params.in_nseg = 1;

	params.out_bdev = dst_bdev;
	params.out_segs = &blk_copy_out_seg;
	params.out_nseg = 1;

	/*
	 * Synchronous operation.
	 */
	params.end_io = NULL;
	params.private = NULL;

	f2fs_info(sbi,
		"[F2FS_GC][FLOW=%016llx] "
		"COPY_OFFLOAD_PREPARED "
		"SRC_PBA=0x%llx SRC_SECTOR=%llu "
		"DST_PBA=0x%llx DST_SECTOR=%llu "
		"LEN=%u "
		"SRC_MAX_COPY=%u DST_MAX_COPY=%u",
		(unsigned long long)flow_id,
		(unsigned long long)old_blkaddr,
		(unsigned long long)src_sector,
		(unsigned long long)new_blkaddr,
		(unsigned long long)dst_sector,
		F2FS_BLKSIZE,
		bdev_max_copy_sectors(src_bdev),
		bdev_max_copy_sectors(dst_bdev));

	if (!bdev_max_copy_sectors(src_bdev) ||
	    !bdev_max_copy_sectors(dst_bdev)) {

		f2fs_info(sbi,
			"[F2FS_GC][FLOW=%016llx] "
			"COPY_OFFLOAD_UNSUPPORTED "
			"SRC_MAX_COPY=%u DST_MAX_COPY=%u",
			(unsigned long long)flow_id,
			bdev_max_copy_sectors(src_bdev),
			bdev_max_copy_sectors(dst_bdev));

		return -EOPNOTSUPP;
	}

	f2fs_info(sbi,
		"[F2FS_GC][FLOW=%016llx] "
		"COPY_OFFLOAD_ATTEMPT "
		"SRC_PBA=0x%llx DST_PBA=0x%llx "
		"LEN=%u",
		(unsigned long long)flow_id,
		(unsigned long long)old_blkaddr,
		(unsigned long long)new_blkaddr,
		F2FS_BLKSIZE);

	ret = blkdev_copy_offload(&params);

	f2fs_info(sbi,
		"[F2FS_GC][FLOW=%016llx] "
		"COPY_OFFLOAD_RETURN "
		"ret=%d SRC_PBA=0x%llx DST_PBA=0x%llx",
		(unsigned long long)flow_id,
		ret,
		(unsigned long long)old_blkaddr,
		(unsigned long long)new_blkaddr);

	return ret;
}
